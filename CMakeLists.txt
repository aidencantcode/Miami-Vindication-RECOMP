cmake_minimum_required(VERSION 3.25)

# Dependencies come from vcpkg (manifest: vcpkg.json). CMakePresets.json points
# CMAKE_TOOLCHAIN_FILE at Visual Studio's bundled vcpkg.
project(GangstarNative LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")

find_package(SDL2 CONFIG REQUIRED)
find_package(OpenGL REQUIRED)
find_package(imgui CONFIG REQUIRED)
find_path(STB_INCLUDE_DIR stb_vorbis.c)  # vcpkg "stb" (header-only)
# Unicorn: the same 2.1.4 build the Python probe uses, prepared by tools/setup_unicorn.ps1
# (headers + import lib + DLL under third_party/unicorn, git-ignored).
set(UNICORN_ROOT "${CMAKE_SOURCE_DIR}/third_party/unicorn" CACHE PATH "Prebuilt Unicorn")
add_library(unicorn SHARED IMPORTED)
set_target_properties(unicorn PROPERTIES
    IMPORTED_IMPLIB "${UNICORN_ROOT}/lib/unicorn.lib"
    IMPORTED_LOCATION "${UNICORN_ROOT}/bin/unicorn.dll"
    INTERFACE_INCLUDE_DIRECTORIES "${UNICORN_ROOT}/include")
set(UNICORN_TARGET unicorn)

add_executable(gangstar
    src/main.cpp
    src/core/log.cpp
    src/core/guest_memory.cpp
    src/loader/elf_image.cpp
    src/hle/imports.cpp
    src/hle/bionic_data.cpp
    src/hle/libc_core.cpp
    src/hle/libc_stdio.cpp
    src/hle/overrides.cpp
    src/cpu/cpu.cpp
    src/rt/runtime.cpp
    src/jni/jni.cpp
    src/gl/gles1.cpp
    src/gl/post.cpp
    src/gl/ffp.cpp
    src/gl/s3tc.cpp
    src/input/input.cpp
    src/settings/settings.cpp
    src/game/mods.cpp
    src/game/night_textures.cpp
    src/ui/overlay.cpp
    src/audio/mixer.cpp
    src/audio/media_player.cpp
    src/game.cpp
    src/selftest.cpp
)
target_include_directories(gangstar PRIVATE src ${STB_INCLUDE_DIR})
target_compile_definitions(gangstar PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
    target_compile_options(gangstar PRIVATE /W4 /permissive- /utf-8 /wd4100)
endif()
target_link_libraries(gangstar PRIVATE SDL2::SDL2 SDL2::SDL2main OpenGL::GL imgui::imgui ${UNICORN_TARGET})

# Copy runtime DLLs next to the exe.
add_custom_command(TARGET gangstar POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_RUNTIME_DLLS:gangstar> $<TARGET_FILE_DIR:gangstar>
    COMMAND_EXPAND_LISTS)

# stb_vorbis is compiled inside mixer.cpp; silence its code-generation warnings.
if(MSVC)
    set_source_files_properties(src/audio/mixer.cpp PROPERTIES COMPILE_OPTIONS "/wd4701;/wd4703")
endif()

# ---- Phase 4: the recompiled game code (generated by tools/recomp/codegen.py into work/recomp,
# git-ignored because it is derived from the game). Strict IEEE floating point so float maths
# matches the phone's soft-float results.
option(GANGSTAR_RECOMP "Build the recompiled game code from work/recomp" OFF)
if(GANGSTAR_RECOMP)
    file(GLOB RECOMP_SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/work/recomp/*.c")
    if(NOT RECOMP_SOURCES)
        message(FATAL_ERROR "work/recomp is empty: run python -I tools/recomp/codegen.py first")
    endif()
    add_library(game_recomp STATIC ${RECOMP_SOURCES})
    target_include_directories(game_recomp PUBLIC "${CMAKE_SOURCE_DIR}/src/recomp" "${CMAKE_SOURCE_DIR}/work/recomp")
    set_target_properties(game_recomp PROPERTIES C_STANDARD 11)
    if(MSVC)
        target_compile_options(game_recomp PRIVATE /O2 /fp:precise /bigobj /W0)
        set_property(TARGET game_recomp PROPERTY MSVC_DEBUG_INFORMATION_FORMAT "")  # 50 MB of C: no PDB
    else()
        target_compile_options(game_recomp PRIVATE -O2 -ffp-contract=off -w)
    endif()
endif()
if(GANGSTAR_RECOMP)
    target_sources(gangstar PRIVATE src/recomp/rc_runtime.cpp)
    if(MSVC)
        # rc_call & co. are extern "C" (called from the generated C) and throw on fatal errors: /EHc-
        # stops MSVC from assuming extern "C" functions never throw.
        set_source_files_properties(src/recomp/rc_runtime.cpp PROPERTIES COMPILE_OPTIONS "/EHc-")
    endif()
    target_link_libraries(gangstar PRIVATE game_recomp)
    target_compile_definitions(gangstar PRIVATE GANGSTAR_HAVE_RECOMP)
    # Every guest call is a host C call in recompiled code: give the main thread a deep stack.
    if(MSVC)
        target_link_options(gangstar PRIVATE /STACK:67108864)
    endif()
endif()
